# 定时任务平台

## 数据库设计

```sql
CREATE TABLE `bs_cron_task` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `name` varchar(64) DEFAULT NULL COMMENT '任务名',
  `desc` varchar(800) DEFAULT NULL COMMENT '任务描述',
  `cron` varchar(32) DEFAULT NULL COMMENT 'cron表达式',
  `ctype` varchar(12) DEFAULT NULL COMMENT '命令类型',
  `command` text DEFAULT NULL COMMENT '命令详情',
  `limit` int DEFAULT NULL COMMENT '执行次数',
  `status` varchar(12) DEFAULT NULL COMMENT '状态',
  `created_at` timestamp NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

INSERT INTO `bs`.`bs_cron_task` (`name`, `desc`, `cron`, `ctype`, `command`, `limit`, `status`, `created_at`, `updated_at`) VALUES ('http_5min', '测试http任务', '*/5 * * * * *', 'http', '{\"ctype\": \"http\",\"method\": \"post\",\"url\": \"http://postman-echo.com/post\",\"header\": {\"auth\": \"xxx-xxx-xxssxx\"},\"body\": \"{\\\"name\\\":\\\"lihua\\\", \\\"age\\\":21}\"}', 99, 'online', '2025-01-13 07:53:12', '2025-01-13 15:53:14');

```

### http命令

```json
{
    "ctype":"http",
    "method":"post",
    "url":"http://postman-echo.com/post",
    "header":{
        "auth":"xxx-xxx-xxssxx"
    },
    "body":"{\"name\":\"lihua\", \"age\":21}"
}
```
